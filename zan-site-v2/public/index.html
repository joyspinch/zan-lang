using Gui;
using Gui.Widget;

class App {
    static void Main() {
        var form  = Form.Create("Zan Demo", 460, 300);
        var label = new Label("点击次数：0");
        var btn   = new Button("点击我");
        int count = 0;

        btn.Click += () => {
            count++;
            label.Text = $"点击次数：{count}";
        };

        form.Add(label, btn).Run();
    }
}

01

下载并安装 ZanIDEDownload & install ZanIDE

Windows / macOS / Linux，双击安装包，首次启动自动配置工具链。 Windows / macOS / Linux — double-click the installer, toolchain configures on first launch.

02

新建项目，选择模板New project, pick a template

03

编写并运行Write & run

F5 直接运行，或 zanc app.zan --auto-stdlib -o app.exe 命令行编译。 Press F5 to run, or compile with zanc app.zan --auto-stdlib -o app.exe on the CLI.

    }
}
using Game;

class App : GameApp {
    Sprite hero;

    override void OnLoad() {
        hero = Sprite.Load("hero.png");
    }

    override void OnUpdate(float dt) {
        if (Input.Right()) hero.X += 120 * dt;
    }

    override void OnDraw() { hero.Draw(); }
}
using Sdk.Http;

class App {
    static void Main() {
        var app = new HttpApp();
        app.Get("/hello", ctx => {
            ctx.Json(new { msg = "hello from Zan" });
        });
        app.Listen(8080);
    }
}
