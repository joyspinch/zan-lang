# Cases for the self-host differential gate (tests/run_selfhost_diff.cmake).
#
# Every case here is a conformance case: the C host compiles it, runs it and
# its stdout is the golden file next to it. The gate requires the self-hosted
# compiler (gen1) to reach the same stdout. A case that still differs is
# listed in diff_known_gaps.txt -- with the reason -- until it is fixed.
#
# Keep this list small and meaningful: each case costs a gen1 compile plus a
# link plus two program runs.

# switch: constant sections, fallthrough into the next section, enum switch
switch_case_fallthrough
switch_break_in_loop
enum_switch
switch_enum_exhaustiveness

# foreach over arrays and string arrays (the indexed walk, not just List<T>)
array_init_foreach

# ordinary statements
const_narrowing
var_call_initializer

# switch patterns and the `is` expression
cs_b05_pattern_var
cs_b06_switch_expr
pattern_value_types

# collections: the enumerable protocol (GetEnumerator/MoveNext/Current) and
# the LINQ iterators built on it
list_wide_struct_elements
enumerable_protocol
value_enumerable
linq_query
list_reserve

# statement shapes and loop-variable capture
stmt_braceless_bodies
for_loopvar_capture

# closure capture semantics (by reference over a written local, the receiver
# of the enclosing method, a delegate call nested in a delegate call)
closure_capture_by_ref
closure_capture_async
closure_captures_this
closure_nested_call
async_boxed_local_release
async_foreach_capture
async_void_call

# string / view semantics
string_view_length
string_nul_search

# the compiler intrinsics the standard library itself calls: the byte[] <-> string
# bridges and the ordinal string compare
bytes_bridge

# a static property is a static getter call (Cls.get_Prop())
static_property

# the builtin surface the self-host lowers itself: the Math family,
# string.Equals, List<T>.ToArray and the Clear methods
builtin_surface

# the Console family: the per-type rendering of Write/WriteLine/PrintLine
# and the ANSI sequences of Clear/ResetColor
console_builtin
console_ansi
console_color

# the task handle family: Task.Run/Spawn typed by the argument shape and the
# no-result handle rejected where a Task<T> is expected (see the gap list)
task_run_action_handle

# the awaited result of a Task-shaped async method: `async Task<T>` carries
# the result, so the awaited value has the element type
async_task_result

# a lambda returns its own value, not the enclosing method's: a block-bodied
# lambda with a value return inside a void method
lambda_value_return

# the type-argument count of a type reference: a built-in generic (one
# argument for List, two for Dictionary/Dict), a user generic class (as many
# arguments as it declares type parameters) and a delegate (Func/Action are
# not counted) must all be accepted
type_arity_ok
generic_arity_check

# the async shapes the C host's parser desugars: `async ValueTask<T>` (the
# name is rewritten, it never reaches the binder) and a Task-returning
# interface method without a body (it is an async method too, so a call
# through the interface is a coroutine call)
async_valuetask
async_interface_task

# a struct is a value: assignment, argument passing and a container element
# all copy it, a nested struct field is copied deeply and a class field is
# shared between the copies
struct_value_copy

# the runtime type test: the built-in containers carry a class id, so
# `o is List<int>` is decided at runtime (and a null container is not one)
runtime_type_test

# the operator overloads: the compound assignment (`+= -= *= /= %=`) goes
# through the op_* method, several overloads of the same operator are picked
# by argument type (`v + 2.5` reaches the double one) and a struct compares
# by value
operator_overload
struct_operators
operator_relational

# the ordinary overload resolution: the candidates come from the whole class
# chain (a derived overload does not hide a base one with a different
# signature), the argument count and the argument types pick the overload, and
# an awaited call's result type is the resolved overload's return type
overload_resolution
overload_inherited_arity
overload_same_arity_return

# the instance field initializers: they run in the order the C host runs them
# (the base class ones before the derived ones, the base constructor in
# between) and they run even when the class has no constructor of its own
field_decl_initializers
constructor_chaining

# the indexer: `obj[i]` reads through the getter, `obj[i] = v` writes through
# the setter, and the compound assignment goes through the setter as well
cs_b09_indexer

# several indexers on one class: the index argument type picks the overload
# (`b[1]` reaches the int one, `b["k"]` the string one)
indexer_overload

# the user-defined conversion operators: `implicit`/`explicit operator T(S)`
# turns a value of type S into one of type T at an assignment, an argument, a
# return and a cast
cs_b12_conv_operator

# a field that hides a base field of the same name: both keep their own slot,
# and the name resolves to the most derived declaration
field_hiding

# a scalar argument for a class parameter goes through the class's
# one-argument constructor (the implicit constructor conversion)
implicit_ctor_argument

# `new List<T>(src)` is the copy constructor: the source's elements are copied
# into the new list, and `new List<T>(a, b)` is a collection initializer
list_copy_ctor

# the invoked-object protocol: `E(args)` on a class-typed local dispatches the
# class's op_call (a static one takes E as its first argument)
cs_b_opcall_params

# the integer width: an int value lives in a 64-bit slot, but a value that
# enters a typed location is narrowed to that type's width (the checked and
# unchecked arithmetic wrap like the host's 32-bit int)
cs_b14_checked
cs_b15_checked_unsigned

# the logical right shift `>>>` (and `>>>=`): the value is widened to 64 bits
# and shifted logically, whatever the operand's signedness
ushr_operator

# the static constructor: `static C() { ... }` runs once at program entry,
# after the class's static field initializers
static_constructor

# a static field initialized with an object (`static object rawLock = new object();`)
static_object_fields

# a field whose name is also a type name: the field wins in an expression
# (`Collapse.Add("x")` with a `Collapse` field is a call on the field)
field_shadows_type

# a null string: the length probe and the element access are soft runtime
# errors -- the program keeps running and reads 0 / NUL
null_string_index

# the raw-memory byte search: NativeMemory.Find measures the hit from the
# base pointer (not from the window start), so a search that resumes at an
# offset finds the byte at its absolute index
bytebuffer_indexof

# a bare name inside a class is the property (get_X/set_X accessor) or the
# field of that class -- an instance one and a static one
bare_name_property

# a type parameter compared against null: the erased slot travels in an i64,
# so the test is against 0, not against the null pointer
generic_type_param_null_check
